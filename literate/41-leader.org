#+TITLE: The leader: real keys under SPC
#+FILETAGS: :keys:
#+STARTUP: overview
#+PROPERTY: header-args:emacs-lisp :tangle ../lisp/kittymacs-leader.el :eval never :comments no :padline no :mkdirp yes

- The smallest and most important module: =SPC= becomes an ordinary prefix keymap in Meow's Normal and Motion states, and every major mode gets its own menu under =SPC m=.
- Knows nothing about other packages or this machine.
  + [[id:53ad53ed-155f-47dc-973d-7be18de2893c][The keys chapter]] fills the tree; package chapters add their =SPC m= menus with one form each.
- Read [[id:a1607a41-b741-428a-9de3-c16e7c8d3ecc][the editing chapter]] first: Meow's states decide when the leader is live.

* Why a real keymap instead of Meow's keypad :decision:
:PROPERTIES:
:ID: fc8a853a-879a-4bc7-81dc-9d2f4ccc5fe2
:END:

- Meow's /keypad/: =SPC= starts a loop that reads keys one at a time and *translates* them (=SPC x f= becomes =C-x f=).
- Clever, but =SPC f f= is then not a binding anywhere, so Emacs's own tools cannot see it:
  + =C-h k= cannot describe it; =C-h b= does not list it;
  + which-key shows the translated keys;
  + =M-x= annotations show the =C-x= form you never type.
- An Emacs keymap is a tree of keys: bind =SPC= to one in the Normal-state map and =SPC f f= is a real binding.
  + Every help command, which-key and Marginalia see exactly the keys you press, with no adapter.
  + Meow's own helper for adding keys to a state accepts a keymap, so the whole leader is two lines.
- The keypad stays available as the command =meow-keypad=, unbound unless =kittymacs-keypad-key= names a key for it.
- The three options:
  + =kittymacs-leader-key= :: the leader, =SPC=;
  + =kittymacs-localleader-key= :: the mode menu below it, =m=;
  + =kittymacs-keypad-key= :: a key below the leader for =meow-keypad=, or =nil=.

#+begin_src emacs-lisp
;;; kittymacs-leader.el --- A literal leader and localleaders for Meow -*- lexical-binding: t; -*-
;; Generated from literate/41-leader.org; edit the Org source, then tangle.

;;; Commentary:
;; SPC becomes an ordinary prefix keymap in Meow's Normal and Motion states,
;; and every major mode gets its own map under SPC m.  Everything is a native
;; keymap, so C-h k, C-h b, which-key and Marginalia see the real keys.

;;; Code:

(declare-function meow-normal-define-key "meow-helpers" (&rest keybinds))
(declare-function meow-motion-define-key "meow-helpers" (&rest keybinds))

(defcustom kittymacs-leader-key "SPC"
  "Key that opens `kittymacs-leader-map' in Meow's Normal and Motion states."
  :type 'key
  :group 'kittymacs)

(defcustom kittymacs-localleader-key "m"
  "Key below the leader that opens the current major mode's localleader map."
  :type 'key
  :group 'kittymacs)

(defcustom kittymacs-keypad-key nil
  "Optional key below the leader that starts Meow's keypad translation loop.
Nil leaves `meow-keypad' unbound; it stays available as a command."
  :type '(choice (const :tag "Unbound" nil) key)
  :group 'kittymacs)

#+end_src

* The leader map

- =kittymacs-leader-map= is a plain keymap:
  + =keymap-set= adds a command;
  + a key bound to a cons of a label and a keymap adds a labelled group, and which-key prints the label instead of "+prefix".
- Both forms are standard Emacs: anything that understands keymaps understands the leader.

#+begin_src emacs-lisp :tangle no
(keymap-set kittymacs-leader-map "f f" #'find-file)
(keymap-set kittymacs-leader-map "f" (cons "files" my-file-map))
#+end_src

#+begin_src emacs-lisp
(defvar kittymacs-leader-map (make-sparse-keymap)
  "The leader map, opened by `kittymacs-leader-key' in Normal and Motion states.
Add a command with `keymap-set'.  Add a labelled group with
  (keymap-set kittymacs-leader-map \"f\" (cons \"files\" my-file-map)).")

#+end_src

* Localleaders: one menu per major mode :concept:
:PROPERTIES:
:ID: 6279ff91-6766-4e41-a4a0-de6e2f9b10ce
:END:

- Evil has per-mode Normal-state keymaps; Meow does not.
- Emacs's own way to get a buffer-specific prefix is =emulation-mode-map-alists=:
  + a list of alists of =(MINOR-MODE-VARIABLE . KEYMAP)=, consulted before minor-mode and major-mode maps;
  + an entry may be a /symbol/ whose value is read in the current buffer, so a buffer-local variable gives each buffer its own map.
- The entry below keys its map on Meow's state variables: the menu appears in Normal and Motion and vanishes in Insert.
- Maps are registered per major mode in =kittymacs-localleader-alist= and composed along the mode's parents, most specific first:
  + a map for =prog-mode= is inherited by every programming mode;
  + a map for =emacs-lisp-mode= overrides the keys it redefines and inherits the rest.
- Why an emulation entry rather than a =menu-item :filter=: only the entry is visible to =where-is=, which Marginalia uses ([[file:../ARCHITECTURE.md][ARCHITECTURE.md]], "Keys and Meow").

#+begin_src emacs-lisp
(defvar kittymacs-localleader-alist nil
  "Alist of (MAJOR-MODE . KEYMAP) localleader maps.
Maps compose along the mode's parents, most specific first, so a map for
`prog-mode' is inherited by every programming mode.")

(defvar-local kittymacs--localleader-alist nil
  "This buffer's emulation-map entry: its composed localleader under the leader.")

(defun kittymacs-localleader-map (&optional mode)
  "Return the composed localleader keymap for MODE, or nil if none applies.
MODE defaults to the current major mode."
  (let ((maps (delq nil (mapcar (lambda (parent)
                                  (alist-get parent kittymacs-localleader-alist))
                                (derived-mode-all-parents (or mode major-mode))))))
    (when maps (make-composed-keymap maps))))

(defun kittymacs--install-localleader ()
  "Point this buffer's emulation entry at its localleader, if it has one."
  (setq kittymacs--localleader-alist
        (when-let* ((local (kittymacs-localleader-map))
                    (wrapper (define-keymap
                               kittymacs-leader-key
                               (define-keymap kittymacs-localleader-key
                                 (cons "mode" local)))))
          `((meow-normal-mode . ,wrapper) (meow-motion-mode . ,wrapper)))))

(defun kittymacs-refresh-localleaders ()
  "Recompute the localleader of every live buffer."
  (dolist (buffer (buffer-list))
    (with-current-buffer buffer (kittymacs--install-localleader))))

(defun kittymacs-define-localleader (mode &rest definitions)
  "Give MODE a localleader built from DEFINITIONS and refresh live buffers.
DEFINITIONS are `define-keymap' arguments: KEY DEFINITION pairs, where a
definition may be (cons \"label\" COMMAND) to label it for which-key.
Calling this again for MODE replaces its map."
  (declare (indent 1))
  (setf (alist-get mode kittymacs-localleader-alist) (apply #'define-keymap definitions))
  (kittymacs-refresh-localleaders)
  mode)

#+end_src

- Defining a localleader is one form per mode, written next to its package: [[id:f6ccd2da-4b3f-4392-b439-3966fc9d3db8][version control]] defines Magit's menu, [[id:8ff7e58e-c155-4513-99ae-3b2c3a6ab729][Org]] Org's.

#+begin_src emacs-lisp :tangle no
(kittymacs-define-localleader 'emacs-lisp-mode
  "e" (cons "eval buffer" #'eval-buffer)
  "d" (cons "edebug defun" #'edebug-defun))
#+end_src

* Turning it on, and seeing what you have

- =kittymacs-leader-enable= binds the leader in both states, registers the localleader entry once and refreshes every buffer.
  + Safe to call again; [[id:53ad53ed-155f-47dc-973d-7be18de2893c][the keys chapter]] calls it once the tree is built.
- =kittymacs-describe-leader= opens Emacs's help view for the leader plus this buffer's localleader.
- =C-h k SPC f f= and which-key need no help from this module.

#+begin_src emacs-lisp
(defun kittymacs-describe-leader ()
  "Describe the leader bindings, including this buffer's localleader."
  (interactive)
  (let ((map (make-sparse-keymap)))
    (set-keymap-parent map kittymacs-leader-map)
    (when-let* ((local (kittymacs-localleader-map)))
      (keymap-set map kittymacs-localleader-key (cons "mode" local)))
    (describe-keymap map)))

(defun kittymacs-leader-enable ()
  "Bind the leader in Meow's Normal and Motion states and start localleaders.
Safe to call more than once."
  (require 'meow)
  (meow-normal-define-key (cons kittymacs-leader-key kittymacs-leader-map))
  (meow-motion-define-key (cons kittymacs-leader-key kittymacs-leader-map))
  (when kittymacs-keypad-key
    (keymap-set kittymacs-leader-map kittymacs-keypad-key #'meow-keypad))
  (add-to-list 'emulation-mode-map-alists 'kittymacs--localleader-alist)
  (add-hook 'after-change-major-mode-hook #'kittymacs--install-localleader)
  (kittymacs-refresh-localleaders))

(provide 'kittymacs-leader)
;;; kittymacs-leader.el ends here
#+end_src

* Check

- =SPC= in a Normal-state buffer: which-key lists the groups after a pause.
- =C-h k SPC f f=: Emacs describes =find-file=, the command the key really runs.
- =SPC m= in an Org buffer, then in an Emacs Lisp buffer: two different menus.
- =i= then =SPC=: Insert state types a space; the leader is gone.
- [[file:../tests/kittymacs-leader-tests.el][The leader tests]] load only Meow and this module and check the promises above:
  + =SPC= is a prefix in Normal and Motion, and a space in Insert;
  + localleaders override and inherit along the mode lineage, and other buffers see nothing;
  + the keypad is unbound unless =kittymacs-keypad-key= is set.
