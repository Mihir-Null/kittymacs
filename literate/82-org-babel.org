- The =org-babel= feature supplies interactive execution and authoring commands.
- Use =:eval never-export= on examples that should export their saved results without running.
- The configuration builder has separate generation declarations.

* Execution and authoring
:PROPERTIES:
:KITTYMACS_GENERATE: ../lisp/kittymacs-org-babel.el
:header-args:emacs-lisp: :tangle ../lisp/kittymacs-org-babel.el :eval never :comments no :padline no :mkdirp yes
:END:

- Core blocks use Emacs Lisp, Bash, Python, and JavaScript. Guile uses =ob-scheme= when selected.
- Optional adapters use the project runtime on the buffer's PATH. Templates are available through Org Tempo.
- Mermaid needs =mmdc= and its browser runtime. Give rendered diagrams an explicit =:file= path.
- Jupyter uses the existing kernel feature. Discover project kernels with =SPC m J=, then use a named session and an explicit =jupyter-python= block.
- Shell sessions have native async execution. The =babel-async= feature handles standalone shell and JavaScript blocks explicitly marked =:async yes=.
- For project code that you intend to detangle, use =:comments link= and review the source diff before saving.

(defun kittymacs-babel--enable (language)
  "Load the backend for LANGUAGE unless it was explicitly disabled."
  (let ((setting (assq language org-babel-load-languages)))
    (when (or (not setting) (cdr setting))
      (condition-case err
          (progn
            (require (intern (format "ob-%s" language)))
            (unless setting
              (setq org-babel-load-languages
                    (append org-babel-load-languages (list (cons language t)))))
            t)
        (error (display-warning 'kittymacs (error-message-string err)))))))

(defun kittymacs-babel--setup ()
  "Set up the interactive Babel backends and editing modes."
  (require 'ob-tangle)
  (dolist (entry '(("bash" sh "sh") ("javascript" js "js")
                   ("python" python "py") ("nix" nix "nix")
                   ("racket" racket "rkt") ("scheme" scheme "scm")
                   ("mermaid" mermaid "mmd") ("cmake" cmake-ts "cmake")
                   ("css" css "css") ("dockerfile" dockerfile-ts "Dockerfile")
                   ("go" go-ts "go") ("gomod" go-mod-ts "mod")
                   ("html" html "html") ("json" json-ts "json")
                   ("markdown" markdown-ts "md") ("rust" rust-ts "rs")
                   ("toml" toml-ts "toml") ("tsx" tsx-ts "tsx")
                   ("typescript" typescript-ts "ts") ("ts-node" typescript-ts "ts")
                   ("yaml" yaml-ts "yaml")))
    (pcase-let ((`(,language ,mode ,extension) entry))
      (when (and (fboundp (intern (format "%s-mode" mode)))
                 (not (assoc language org-src-lang-modes)))
        (push (cons language mode) org-src-lang-modes))
      (unless (assoc language org-babel-tangle-lang-exts)
        (push (cons language extension) org-babel-tangle-lang-exts))))
  (mapc #'kittymacs-babel--enable '(emacs-lisp shell python js))
  (dolist (entry '((lang-guile . scheme) (babel-nix . nix)
                   (babel-racket . racket) (babel-go . go) (babel-rust . rust)
                   (babel-typescript . ts-node) (babel-mermaid . mermaid)))
    (when (kittymacs-feature-p (car entry))
      (kittymacs-babel--enable (cdr entry)))))

(defun kittymacs-babel--execute (original &optional arg info params executor-type)
  "Route a standalone async block to ob-async, preserving native dispatch."
  (let* ((probe (or info (org-babel-get-src-block-info 'light)))
         (language (car probe))
         (headers (and probe (org-babel-merge-params (nth 2 probe) params))))
    (if (and (kittymacs-feature-p 'babel-async)
             (memq executor-type '(nil src-block))
             (member language '("bash" "sh" "shell" "js"))
             (equal (alist-get :async headers) "yes")
             (member (alist-get :session headers) '(nil "none"))
             (not (file-remote-p default-directory))
             (require 'ob-async nil t))
        (let ((org-babel-load-languages
               (list (cons (if (equal language "js") 'js 'shell) t))))
          (ob-async-org-babel-execute-src-block original arg info params))
      (if executor-type
          (funcall original arg info params executor-type)
        (funcall original arg info params)))))

(with-eval-after-load 'ob-async
  (advice-remove 'org-babel-execute-src-block #'ob-async-org-babel-execute-src-block))

(defun kittymacs-babel-jupyter-refresh ()
  "Discover Jupyter languages from the current project's kernels."
  (interactive)
  (when (kittymacs-babel--enable 'jupyter)
    (org-babel-jupyter-aliases-from-kernelspecs t)))

(with-eval-after-load 'org
  (kittymacs-babel--setup)
  (when (kittymacs-feature-p 'babel-async)
    (advice-add 'org-babel-execute-src-block :around #'kittymacs-babel--execute))
  (when (kittymacs-feature-p 'babel-jupyter)
    (kittymacs-babel--enable 'jupyter)))
(with-eval-after-load 'org-tempo
  (dolist (entry (append '(("sh" . "src bash") ("py" . "src python")
                           ("js" . "src js"))
                         (when (kittymacs-feature-p 'babel-mermaid)
                           '(("mm" . "src mermaid")))
                         (when (kittymacs-feature-p 'babel-jupyter)
                           '(("jp" . "src jupyter-python")))))
    (unless (assoc (car entry) org-structure-template-alist)
      (push entry org-structure-template-alist))))

(let ((map (alist-get 'org-mode kittymacs-localleader-alist)))
  (when map
    (keymap-set map "B" (cons "tangle current document" #'org-babel-tangle))
    (when (kittymacs-feature-p 'babel-jupyter)
      (keymap-set map "J" (cons "discover Jupyter kernels" #'kittymacs-babel-jupyter-refresh)))
    (kittymacs-refresh-localleaders)))
