#+TITLE: Platform: paths, shells, spelling, macOS and Android
#+FILETAGS: :core:
#+STARTUP: overview
#+PROPERTY: header-args:emacs-lisp :tangle ../lisp/kittymacs-platform.el :eval never :comments no :padline no :mkdirp yes

- Answers one question for every later chapter: /where does this machine keep things, and what is it allowed to assume?/
  + Emacs core supplies process creation and shell variables; this module discovers paths, picks the system shell, finds the spell checker and applies the macOS and Android policy.
  + It prefers discovery (=executable-find=, the platform's own home) over machine-specific paths.
- Leans on [[id:40302e33-d2b2-4c0b-9bc3-1f9e121307e1][startup]]: =init.el= loads =private.el= right after this module, then calls =kittymacs-platform-apply=.
- Must not declare a package: [[file:../tests/kittymacs-platform-tests.el][the platform tests]] load it with none, and run every operating system's branch on any machine by binding =system-type=.
  + Nor does it force a font: inheriting the platform default makes first boot robust; [[id:0286f652-e861-4e61-9ca6-91e225c8939d][the appearance chapter]] chooses fonts.
- PowerShell stays the ordinary Windows shell; MSYS2 UCRT64 is a separate choice offered by [[id:bd2638bc-735d-4849-bafe-3dab5e8e30b7][the terminal chapter]], not a global environment replacement.

* Where your files live

- The options below are in the =kittymacs-platform= group, so =M-x customize-group= finds them; =private.el= can override any of them.
- Native Windows Emacs may define =HOME= as =AppData/Roaming=; =USERPROFILE= is a better default for your projects and documents.
  + Other platforms keep their normal home.

#+begin_src emacs-lisp
;;; kittymacs-platform.el --- Portable platform defaults -*- lexical-binding: t; -*-
;; Generated from literate/30-platform.org; edit the Org source, then tangle.

;;; Commentary:
;; Safe defaults for Windows, GNU/Linux/Nix, macOS and Android.  Prefer
;; discovery with `executable-find' and platform-provided home locations over
;; machine-specific paths.  The macOS section is distilled from Lambda-Emacs by
;; Colin McLear (`lem-setup-macos', GPL-3.0-or-later).

;;; Code:

(require 'seq)
(require 'subr-x)
(defgroup kittymacs-platform nil
  "Where this machine keeps things: paths, the shell, spelling, macOS and Android."
  :group 'kittymacs)
(defun kittymacs--user-home-directory ()
  "Return the user's ordinary home directory for configuration defaults.
On native Windows, Emacs may define HOME as AppData/Roaming, so prefer
USERPROFILE for user-owned projects and documents."
  (file-name-as-directory
   (if (eq system-type 'windows-nt)
       (or (getenv "USERPROFILE") (expand-file-name "~"))
     (expand-file-name "~"))))
#+end_src

- =kittymacs-project-directory= :: where to look for projects; =~/Projects/= by default, a portable default rather than a machine-specific path.
  + Set it in =private.el=: the project module reads it as it loads.

#+begin_src emacs-lisp
(defcustom kittymacs-project-directory
  (expand-file-name "Projects/" (kittymacs--user-home-directory))
  "Default place to look for projects."
  :type 'directory)
#+end_src

- =kittymacs-org-directory= :: a small, independent starter Org directory, =~/Documents/org/=; [[id:8ff7e58e-c155-4513-99ae-3b2c3a6ab729][Org]] uses it.
  + Change it explicitly to keep notes elsewhere.

#+begin_src emacs-lisp
(defcustom kittymacs-org-directory
  (expand-file-name "Documents/org/" (kittymacs--user-home-directory))
  "Portable starter Org directory."
  :type 'directory)

* Termux on Android :android:
:PROPERTIES:
:ID: ca1bff94-4ca3-40b4-8cda-6a290e7889a5
:END:

- The Android port is a self-contained application; its home directory is its own data directory.
  + Android 10 and later forbid it from executing anything stored there, so Emacs runs subprocesses through an executable loader.
- It cannot reach /another/ application's data directory at all, yet =git=, =ripgrep=, a spell checker and the language servers live in [[https://termux.dev][Termux]], under a different user.
- The two can share a user when installed from builds whose =sharedUserId= and signature match; [[https://sourceforge.net/projects/android-ports-for-gnu-emacs/files/termux/][Jianwei Hu's Android ports]] publish such a pair.
- So everything here /discovers/ Termux rather than assuming it:
  + =kittymacs-termux-root= :: Termux's root, =TERMUX_ROOT= or =/data/data/com.termux/files/usr/=;
  + =kittymacs-termux-program= :: a Termux program's path, or nil when the directory is unreadable, as in a plain, unpaired installation;
  + =kittymacs-termux-bin-directory= :: Termux's =bin=, or nil;
  + every dependent decision degrades to the Android system's own tools.
- =kittymacs-android-p= is a function rather than a constant so the tests can exercise the Android branches by binding =system-type=.

#+begin_src emacs-lisp
(defun kittymacs-android-p ()
  "Return non-nil when this Emacs is the Android port.
A function rather than a constant so the tests can exercise the Android
branches by binding `system-type' on any machine."
  (eq system-type 'android))

(defcustom kittymacs-termux-root
  (file-name-as-directory
   (or (getenv "TERMUX_ROOT") "/data/data/com.termux/files/usr/"))
  "Root of the Termux installation whose programs Emacs may run on Android.
Reachable only when Emacs and Termux share a user ID; otherwise Android
denies access and every lookup below quietly returns nil."
  :type 'directory)

(defun kittymacs-termux-program (name)
  "Return Termux's NAME program when Emacs is allowed to run it, else nil."
  (let ((program (expand-file-name (concat "bin/" name) kittymacs-termux-root)))
    (and (file-executable-p program) program)))

(defun kittymacs-termux-bin-directory ()
  "Return Termux's program directory, without a trailing slash, or nil."
  (let ((bin (expand-file-name "bin" kittymacs-termux-root)))
    (and (file-accessible-directory-p bin) bin)))
#+end_src

* Finding programs

- =kittymacs--first-executable= tries candidates in order on =exec-path=, so the same configuration uses whichever suitable program a machine has installed.

#+begin_src emacs-lisp
(defun kittymacs--first-executable (&rest programs)
  "Return the first executable found in PROGRAMS."
  (seq-some #'executable-find programs))
#+end_src

* The policy applied after private.el :decision:
:PROPERTIES:
:ID: af9044c2-44a8-4795-9f1b-2c0d83d66d76
:END:

- =kittymacs-platform-apply= runs from =init.el= right after =private.el=, so it sees your overrides.
- It picks the /shell/ for =M-x shell= and for every command Emacs runs through a shell:

| System       | Shell: the first one found                              |
|--------------+---------------------------------------------------------|
| Windows      | =pwsh=, =powershell=, then =cmd.exe= as the last resort |
| Linux, macOS | =zsh=, =bash=, =sh= on =exec-path=                      |
| Android      | Termux's =bash= or =sh=, then =/system/bin/sh=          |

- Windows: PowerShell, the newer =pwsh= when installed.
- Android has no =/bin=, so Emacs's compiled-in =/bin/sh= does not exist.
  + Termux's shells come first because they can see Termux's programs.
  + =/system/bin/sh=, always present, is a real fallback, not a hopeful one.
- =shell-file-name= and =explicit-shell-file-name= are user options, set with =setopt=; =shell-command-switch=, the flag that says "run this command", is a plain variable.
- On Windows it borrows Unix tools, last on =exec-path=: Magit's hunk refinement, Ediff and diff-hl call =diff=, =diff3= and =patch= by name, and Windows has none.
  + They come from Git for Windows (=Git/usr/bin=, or beside =git= itself), else from MSYS2's =usr/bin=.
  + As they go last, every native program still wins; =PATH= is left alone, so subprocesses see no change.
- It hands over to [[*macOS]] and [[*Android]].

#+begin_src emacs-lisp
(defun kittymacs--platform-shell ()
  "Return (PROGRAM . SWITCH) for this platform's shell, or nil if none is found."
  (pcase system-type
    ('windows-nt
     (if-let* ((powershell (kittymacs--first-executable "pwsh.exe" "powershell.exe")))
         (cons powershell "-Command")
       (when-let* ((cmd (executable-find "cmd.exe")))
         (cons cmd "/c"))))
    ((or 'darwin 'gnu/linux)
     (when-let* ((shell (kittymacs--first-executable "zsh" "bash" "sh")))
       (cons shell "-c")))
    ('android
     (cons (or (kittymacs-termux-program "bash")
               (kittymacs-termux-program "sh")
               "/system/bin/sh")
           "-c"))))

(defun kittymacs--windows-unix-tools ()
  "Return the directory holding Git for Windows' or MSYS2's Unix tools, or nil.
Git for Windows ships `diff', `diff3' and `patch' beside its own `git';
MSYS2 keeps them under `usr/bin'."
  (let ((git (file-name-directory (or (executable-find "git") ""))))
    (seq-find (lambda (dir) (file-executable-p (expand-file-name "diff.exe" dir)))
              (delq nil
                    (list
                     ;; git in Git/cmd or Git/bin: the tools are in Git/usr/bin.
                     ;; git found in Git/usr/bin itself: they are beside it.
                     (and git (expand-file-name "../usr/bin" git))
                     git
                     (expand-file-name "usr/bin" kittymacs-msys2-root))))))

(defun kittymacs-platform-apply ()
  "Apply the currently configured portable platform defaults."
  (when-let* ((shell (kittymacs--platform-shell)))
    (setopt shell-file-name (car shell)
            explicit-shell-file-name (car shell))
    (setq shell-command-switch (cdr shell)))
  (when (eq system-type 'windows-nt)
    (when-let* ((tools (kittymacs--windows-unix-tools)))
      (add-to-list 'exec-path tools t)))
  (when (eq system-type 'darwin)
    (kittymacs--platform-apply-macos))
  (when (kittymacs-android-p)
    (kittymacs--platform-apply-android)))
#+end_src

* macOS :macos:
:PROPERTIES:
:ID: 1b791960-ef80-43bb-bbf0-876018f0a2f5
:END:

- Emacs on macOS is the Cocoa ("NS") build: it already binds the everyday Command keys (=⌘C=, =⌘V=, =⌘X=, =⌘A=, =⌘S=, =⌘Z=, =⌘W=, =⌘N=) and opens files from Finder.
- It leaves undecided:
  + which physical keys are Meta and Super;
  + whether deleting a file is permanent;
  + what locale a Dock-launched Emacs has;
  + whether the title bar is light or dark.
- This section decides, following Lambda-Emacs's =lem-setup-macos=.
  + =private.el= can override the modifier keys (=kittymacs-macos-modifiers=) and the locale (=LANG= is set only when unset); the rest is applied unconditionally, after =private.el=.
  + None of it runs on another operating system.

** Modifier keys :decision:
:PROPERTIES:
:ID: db2049ab-348e-42c9-a915-c2134d0e379f
:END:

- Option is Meta, Command is Super, and the right Option key is left to macOS so accented characters still type.
- Lambda also made the Fn key Hyper; not carried over, because it turns =Fn=-arrow paging into Hyper chords.
- =kittymacs-macos-modifiers= pairs each NS modifier variable with =meta=, =super=, =hyper=, =control=, =alt=, or =none= to leave the key to macOS.

#+begin_src emacs-lisp
;; Defined by the Cocoa build and by auth-source; declared so the module
;; byte-compiles cleanly on every platform.
(defvar ns-use-native-fullscreen)
(defvar auth-sources)

(defcustom kittymacs-macos-modifiers
  '((ns-command-modifier . super)
    (ns-option-modifier . meta)
    (ns-right-option-modifier . none))
  "How the macOS modifier keys map to Emacs modifiers.
Each entry pairs an NS modifier variable with the modifier it produces:
`meta', `super', `hyper', `control', `alt', or `none' to leave the key
to macOS.  Applied by `kittymacs-platform-apply' after `private.el'."
  :type '(alist :key-type symbol :value-type symbol))
#+end_src

** Trash, quitting and the title bar

- Deleting a file moves it to the Trash, by the best means available:
  1. this Emacs's own =system-move-file-to-trash=, when it has one;
  2. the =trash= command-line tool (=brew install trash=), which keeps Finder's "Put Back";
  3. otherwise =~/.Trash= directly.
- =⌘Q= closes the frame when others remain (frames are this configuration's windows) and quits Emacs from the last one.
- The title bar takes the theme's light or dark appearance, now and on every theme change.
  + =early-init.el= already made it transparent, before the first frame.

#+begin_src emacs-lisp
(defun kittymacs--macos-trash (path)
  "Move PATH to the macOS Trash with the `trash' command-line tool."
  (let ((status (call-process "trash" nil nil nil (expand-file-name path))))
    (unless (eql status 0)
      (error "Failed to move %s to the Trash (exit code %s)" path status))))

(defun kittymacs--macos-configure-trash ()
  "Send deleted files to the Trash by the best available means.
Return the means chosen: `native' when this Emacs moves files to the Trash
itself, `trash-command' for the `trash' tool, or `directory' for ~/.Trash."
  (setopt delete-by-moving-to-trash t)
  (cond ((fboundp 'system-move-file-to-trash) 'native)
        ((executable-find "trash")
         (setopt trash-directory nil)
         (defalias 'system-move-file-to-trash #'kittymacs--macos-trash)
         'trash-command)
        (t (setopt trash-directory "~/.Trash")
           'directory)))

(defun kittymacs-delete-frame-or-quit ()
  "Close this frame when other frames remain; from the last one, quit Emacs."
  (interactive)
  (if (cdr (frame-list))
      (delete-frame)
    (save-buffers-kill-emacs)))

(defun kittymacs--macos-sync-titlebar (&rest _)
  "Give every frame's title bar the theme's light or dark appearance."
  (when (display-graphic-p)
    (let* ((background (face-background 'default nil (selected-frame)))
           (rgb (and (stringp background) (color-name-to-rgb background)))
           (appearance (if (and rgb (color-dark-p rgb)) 'dark 'light)))
      (setf (alist-get 'ns-appearance default-frame-alist) appearance)
      (dolist (frame (frame-list))
        (when (display-graphic-p frame)
          (set-frame-parameter frame 'ns-appearance appearance))))))
#+end_src

** Applying the macOS policy

- Runs from =kittymacs-platform-apply=, on macOS only.
- Full screen is non-native, so a full-screen frame does not move to its own Space and leave every new frame behind.
- Subprocesses get a UTF-8 locale when the Dock started Emacs without one.
- Passwords in Keychain become visible to =auth-source=, which Magit, Forge and Tramp consult.
- Three key bindings fill gaps in the NS port's own:

| Key     | Does                     |
|---------+--------------------------|
| =⌘⇧Z=   | redo                     |
| =⌘Q=    | close frame, or quit     |
| =C-⌘F=  | toggle full screen       |

- Only the Keychain waits for a library, and it waits outside the policy function:
  + =with-eval-after-load= registers its body permanently, once per call: in a function that may run more than once (the tests call =kittymacs-platform-apply= dozens of times) it would queue the same work again on every call;
  + registered once when the module loads, it asks which platform it is on when =auth-source= arrives.

#+begin_src emacs-lisp
(defun kittymacs--platform-apply-macos ()
  "Apply the macOS policy: modifiers, Trash, locale, Keychain, keys, title bar."
  (pcase-dolist (`(,variable . ,modifier) kittymacs-macos-modifiers)
    (set variable modifier))
  (setopt ns-use-native-fullscreen nil)
  (unless (getenv "LANG")
    (setenv "LANG" "en_US.UTF-8"))
  (kittymacs--macos-configure-trash)
  (keymap-global-set "s-Z" #'undo-redo)
  (keymap-global-set "s-q" #'kittymacs-delete-frame-or-quit)
  (keymap-global-set "C-s-f" #'toggle-frame-fullscreen)
  (add-hook 'enable-theme-functions #'kittymacs--macos-sync-titlebar)
  (kittymacs--macos-sync-titlebar))

(defun kittymacs--macos-auth-sources ()
  "Let `auth-source' read passwords from the macOS Keychain."
  (dolist (source '(macos-keychain-internet macos-keychain-generic))
    (add-to-list 'auth-sources source t)))

(with-eval-after-load 'auth-source
  (when (eq system-type 'darwin)
    (kittymacs--macos-auth-sources)))
#+end_src

* Android :android:
:PROPERTIES:
:ID: 6c4986fa-e9ba-4857-bc91-8f59a906d57e
:END:

- Emacs on Android is a real GUI Emacs, not a terminal in a shell app: frames are graphical, fonts are fonts, and =display-graphic-p= is true.
- Three things about the platform still reach up into this configuration:
  1. programs live in another application: [[*Termux on Android]], discovered and optional;
  2. the volume keys are Emacs's escape hatch;
  3. virtual keyboards edit the buffer directly ([[*Virtual keyboards and Meow]]).
- Without a physical keyboard there is no =C-g=, so the Android port reserves the volume keys: two rapid presses of volume down quit.
  + With a keyboard attached that is mostly a nuisance, so =kittymacs-android-volume-keys= can hand them back to the system.
  + The default is Emacs's own behaviour: losing the only way to interrupt a runaway command is a poor thing to discover by accident.
- A physical keyboard reports one irregularity: Android swaps =Alt= and =Meta=, so the =Alt= key produces Emacs's =Meta= modifier, which is what you want anyway.
  + =Super= is labelled =SYM=.
  + =C-SPC= is intercepted before input methods can swallow it, so Meow's leader and =set-mark-command= both survive.
- The two options:
  + =kittymacs-android-volume-keys= :: nil (default) leaves the volume keys to Emacs; non-nil returns them to the system;
  + =kittymacs-android-modal-text-conversion= :: t (default) makes the on-screen keyboard follow Meow's Insert state; nil leaves the input method alone.

#+begin_src emacs-lisp
;; Defined by the Android build; declared so the module byte-compiles cleanly
;; on every platform, as the NS variables above are.
(defvar android-pass-multimedia-buttons-to-system)
(defvar text-conversion-style)

(defcustom kittymacs-android-volume-keys nil
  "Who receives the volume keys on Android.
Nil leaves them to Emacs, which quits on two rapid presses of volume
down and is the only way to interrupt Emacs without a physical keyboard.
Non-nil returns them to the system so they adjust the volume."
  :type 'boolean)

(defcustom kittymacs-android-modal-text-conversion t
  "Whether Android's on-screen keyboard follows Meow's Insert state.
Android input methods edit the buffer directly rather than sending keys.
That suits Insert state and corrupts every other one, so when this is
non-nil the conversion style is suspended outside Insert state.  Set it
to nil to leave the input method alone."
  :type 'boolean)
#+end_src

** Virtual keyboards and Meow :concept:
:PROPERTIES:
:ID: 1c7f1e2c-156b-4aa6-ae7f-cf663018ab44
:END:

- The one real conflict with Meow.
- A desktop input method composes text, then sends key events; an Android one performs /text conversion/, editing the selected window itself.
- Emacs enables it wherever the buffer-local =text-conversion-style= is non-nil: every =text-mode= and =prog-mode= derivative.
  + Exactly right in Meow's Insert state; a disaster in Normal state, where autocorrect and suggestions rewrite a buffer whose keys are commands.
- So the style is suspended on leaving Insert state and restored on re-entry.
  + Emacs rereads the variable only when a buffer becomes selected, a whole state transition too late, so the change goes through =set-text-conversion-style=.
  + Its documentation warns that resetting an input method is not cheap: hence a per-state action, not a per-command one, and =kittymacs-android-modal-text-conversion= to switch it off entirely if a particular keyboard misbehaves.
- [[id:a1607a41-b741-428a-9de3-c16e7c8d3ecc][The editing chapter]] explains Meow's states.

#+begin_src emacs-lisp
(defvar-local kittymacs--android-text-conversion nil
  "The `text-conversion-style' suspended when Meow left Insert state.")

(defun kittymacs--android-suspend-text-conversion ()
  "Stop the on-screen keyboard editing this buffer outside Insert state."
  (when (and kittymacs-android-modal-text-conversion
             (fboundp 'set-text-conversion-style)
             (bound-and-true-p text-conversion-style))
    (setq kittymacs--android-text-conversion text-conversion-style)
    (set-text-conversion-style nil)))

(defun kittymacs--android-resume-text-conversion ()
  "Give the on-screen keyboard this buffer back on entering Insert state."
  (when (and kittymacs-android-modal-text-conversion
             (fboundp 'set-text-conversion-style)
             kittymacs--android-text-conversion)
    (set-text-conversion-style kittymacs--android-text-conversion)
    (setq kittymacs--android-text-conversion nil)))
#+end_src

** Applying the Android policy

- Runs from =kittymacs-platform-apply=, on Android only.
- Termux's =bin=, when reachable, goes first on =PATH= and =exec-path=.
- =LD_LIBRARY_PATH= is cleared, not set.
  + Termux's programs record where their own shared libraries are; an inherited library path interposes Android's same-named system libraries ahead of Termux's, with failures that look like nothing at all.
- =LANG= becomes =en_US.UTF-8= when unset.
- The volume keys go where =kittymacs-android-volume-keys= says; Meow's Insert hooks suspend and resume text conversion.

#+begin_src emacs-lisp
(defun kittymacs--platform-apply-android ()
  "Apply the Android policy: Termux tools, locale, volume keys, keyboard."
  (when-let* ((bin (kittymacs-termux-bin-directory)))
    (unless (member bin (split-string (or (getenv "PATH") "") path-separator t))
      (setenv "PATH" (concat bin path-separator (getenv "PATH"))))
    (add-to-list 'exec-path bin)
    ;; Termux's programs know where their own libraries are.  An inherited
    ;; library path puts Android's system libraries of the same names first
    ;; and breaks them in ways that are very hard to read.
    (setenv "LD_LIBRARY_PATH" nil))
  (unless (getenv "LANG")
    (setenv "LANG" "en_US.UTF-8"))
  (setopt android-pass-multimedia-buttons-to-system kittymacs-android-volume-keys)
  (add-hook 'meow-insert-exit-hook #'kittymacs--android-suspend-text-conversion)
  (add-hook 'meow-insert-enter-hook #'kittymacs--android-resume-text-conversion))
#+end_src

* Revealing a file in the file manager

- One command for every platform, =kittymacs-reveal-in-file-manager=:

| System    | Opens                                                        |
|-----------+--------------------------------------------------------------|
| macOS     | Finder, with the file selected (=open -R=)                   |
| Windows   | Explorer, with the file selected                             |
| Android   | Dired, the file manager that is actually there               |
| elsewhere | the desktop's default handler for the directory (=xdg-open=) |

- Android has no =xdg-open=, and its document picker addresses files by content URI rather than by path, so there is nothing to hand a file manager.
- It takes the buffer's file, the file at point in Dired, or the current directory.
- [[id:53ad53ed-155f-47dc-973d-7be18de2893c][The keys chapter]] binds it to =SPC f o=.

#+begin_src emacs-lisp
(defun kittymacs-reveal-in-file-manager (&optional file)
  "Show FILE in the desktop file manager, selected where the manager allows.
FILE defaults to this buffer's file, the file at point in Dired, or the
current directory."
  (interactive)
  (let* ((file (expand-file-name
                (or file
                    buffer-file-name
                    (and (derived-mode-p 'dired-mode)
                         (fboundp 'dired-get-filename)
                         (dired-get-filename nil t))
                    default-directory)))
         (directory (if (file-directory-p file)
                        (file-name-as-directory file)
                      (file-name-directory file))))
    (pcase system-type
      ('darwin (call-process "open" nil 0 nil "-R" file))
      ;; Explorer wants backslashes; `convert-standard-filename' would do
      ;; it, but only on a Windows build, and the tests run this branch
      ;; anywhere by binding `system-type'.
      ('windows-nt (call-process "explorer.exe" nil 0 nil
                                 (concat "/select," (subst-char-in-string ?/ ?\\ file))))
      ;; Android has no `xdg-open', and its document picker addresses files
      ;; by content URI rather than by path, so there is nothing to hand a
      ;; file manager.  Dired is the file manager that is actually here.
      ('android (dired directory))
      (_ (call-process "xdg-open" nil 0 nil directory)))))

* Spell checking

- Which spell checker exists is a property of the machine, so the search happens here: =hunspell= or =aspell= on =exec-path=, or MSYS2's =hunspell= on Windows.
  + Nothing is enabled when none is installed.
  + A checker without a dictionary reports once and is switched off rather than breaking every buffer.
  + =SPC t s= toggles it.
- Ispell is configured in one guarded function, when Ispell loads, which sets the environment Hunspell needs before pointing Ispell at it.
  + =setopt ispell-program-name= runs a setter that asks Hunspell to list its dictionaries, and Hunspell needs a default dictionary name from the environment even for that; Windows sets no =LANG=.
  + So =DICTIONARY= (=en_US= unless already set) and, for MSYS2's Hunspell, =DICPATH= come first.
  + Guarded because an error here aborts the load of Ispell itself: Emacs restores Ispell's autoload stubs, and every later call reloads the library, runs the checker and fails again.
- The hooks that turn Flyspell on are added unconditionally; each asks for a checker only when a buffer opens.
  + Deciding once, at load time, would be too early: the module loads before =private.el=, so a =kittymacs-msys2-root= set there, or a checker that =exec-path-from-shell= later finds on the login shell's =PATH=, would never turn spelling on.
  + The question costs two lookups on =exec-path= per buffer, nothing next to what Flyspell itself does.
  + Text modes get =flyspell-mode=; programming modes =flyspell-prog-mode=, which checks only comments and strings.
- Only Emacs's own Flyspell is used here; the two front ends that make corrections pleasant (=C-;= and =SPC s p=) are declared in [[id:42b02bb4-ea68-4981-b427-35705bf162f4][the completion chapter]].

#+begin_src emacs-lisp
(defun kittymacs-spell-checker ()
  "Return the spell-checker program to use, or nil."
  (or (executable-find "hunspell")
      (executable-find "aspell")
      (let ((msys2 (expand-file-name "ucrt64/bin/hunspell.exe" kittymacs-msys2-root)))
        (and (eq system-type 'windows-nt) (file-executable-p msys2) msys2))))

;; A checker that exists but has no dictionary must never break anything:
;; say so once and carry on.
(defvar kittymacs--spell-warned nil)
(defun kittymacs--spell-off (err)
  "Report once that spell checking is off because of ERR."
  (unless kittymacs--spell-warned
    (setq kittymacs--spell-warned t)
    (message "Spell checking off: %s" (error-message-string err))))

(defun kittymacs--configure-ispell ()
  "Point Ispell at the machine's checker; runs once, when Ispell loads.
Setting `ispell-program-name' with `setopt' runs its setter, which asks
Hunspell to list its dictionaries, and Hunspell needs a default
dictionary name from the environment even for that; Windows sets no
LANG.  So the environment comes first.  The whole thing is guarded
because an error here aborts the load of Ispell itself: Emacs then
restores Ispell's autoload stubs, and every later call reloads the
library, runs the checker and fails again."
  (when-let* ((program (kittymacs-spell-checker)))
    (condition-case err
        (progn
          (when (string-match-p "hunspell" program)
            (unless (getenv "DICTIONARY") (setenv "DICTIONARY" "en_US"))
            (when (string-prefix-p (expand-file-name kittymacs-msys2-root) program)
              (setenv "DICPATH" (expand-file-name "ucrt64/share/hunspell" kittymacs-msys2-root)))
            (setopt ispell-dictionary "en_US"))
          (setopt ispell-program-name program))
      (error (kittymacs--spell-off err)))))

(with-eval-after-load 'ispell (kittymacs--configure-ispell))

;; Enable Flyspell the same way: on any error say so once and carry on.
(defun kittymacs--flyspell (mode-function)
  "Enable Flyspell with MODE-FUNCTION when this machine has a spell checker.
A checker that fails is reported once instead of breaking the buffer."
  (when (kittymacs-spell-checker)
    (condition-case err
        (funcall mode-function)
      (error (kittymacs--spell-off err)))))
(defun kittymacs--flyspell-text () (kittymacs--flyspell #'flyspell-mode))
(defun kittymacs--flyspell-prog () (kittymacs--flyspell #'flyspell-prog-mode))

(add-hook 'text-mode-hook #'kittymacs--flyspell-text)
(add-hook 'prog-mode-hook #'kittymacs--flyspell-prog)

- =C-h v shell-file-name=: this machine's shell, as in [[*The policy applied after private.el]].
- =SPC f o= in a file buffer: the file manager opens on it, selected where the manager allows.
- =SPC t s= in a text buffer: misspellings are underlined when a checker is installed; without one, nothing happens and nothing breaks.

- Android: two quick presses of volume down quit; in Normal state the on-screen keyboard no longer rewrites the buffer.

  + Git's Unix tools last on =exec-path= on Windows, and nothing changed without them;
  + Termux's shell first, the system shell without it, text conversion following Meow;
  + the file manager for each desktop;
  + Ispell told its dictionary before it is asked, a broken checker reported once, never aborting Ispell.
