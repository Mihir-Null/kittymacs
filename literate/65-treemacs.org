#+TITLE: Treemacs: the project tree
#+FILETAGS: :apps:
#+STARTUP: overview
#+PROPERTY: header-args:emacs-lisp :tangle ../lisp/kittymacs-treemacs.el :eval never :comments no :padline no :mkdirp yes

- =SPC t d= opens a sidebar down the left of the frame showing the current project's directory structure.
  + =j= and =k= move, =TAB= opens a directory, =RET= opens a file.
  + =SPC o t= jumps into it, =q= leaves it, =SPC f t= reveals the file you are editing.
  + It follows you: change to a buffer from another project and the tree switches to that project, unasked.
  + Files are coloured by their Git status; the tree notices changes on disk, including Magit's when you stage or commit.
- [[https://github.com/Alexander-Miller/treemacs][Treemacs]] by Alexander Miller is the file tree; this chapter scopes it to the current project, points its state files at =var/=, and gives it the icons, Git colours and Meow state the rest of this configuration uses.
- It is a /panel/, not a buffer you edit, so it stays inside the current frame instead of becoming its own OS window; [[id:f1d045c2-e2dc-4514-ab17-dd982ed6be00][the frames chapter]] explains the distinction.
- Leans on [[id:0286f652-e861-4e61-9ca6-91e225c8939d][the appearance chapter]] for icons, [[id:e4d65984-892d-4046-b718-d4265ab74215][navigation]] for workspaces and [[id:878a9e46-62be-4b9f-82f6-589560d2ebc1][Dired]] for operating on files.

* The sidebar

- Everything Treemacs writes goes under =var/=, like every other package here.
  + Otherwise it would create =.cache/= beside =init.el=, which Git does not ignore.
  + The workspace layout is state, worth keeping between sessions, so it goes under =var/etc/=; the error file goes to the cache.
- The sidebar is 35 columns and stays put; =SPC m w= sets a width and Treemacs's own =t w= unlocks dragging.
- =treemacs-is-never-other-window= keeps =other-window=, =ace-window= and =C-x o= out of the tree.
  + A frame here normally holds one buffer and the tree, so without it every "other window" lands in the sidebar.
- Single-child directories collapse into one line, so a deep package path does not cost five rows.
  + This needs Python 3; without it the tree shows every level.
- Always on: filesystem watching, a fringe mark on the current file, indent guides, and following the project you are in and only that project.
- Git status colours:
  + with Python 3, the /deferred/ variant: directories as well as files, the status parsed asynchronously, and the tree drawn first and coloured a moment later, which feels quicker on a large repository;
  + without Python 3, the /simple/ variant colours files only;
  + without =git=, no colours.
- The sidebar is a panel that serves the buffer beside it, so =dimmer= leaves it alone: dimming it as an "inactive window" only makes the tree harder to read.
- PNG icons are off in a text terminal.

;; current buffer, with Git status, Nerd Font icons, per-workspace scoping and
;; Magit-aware refreshes.

(require 'kittymacs-leader)

#+begin_src emacs-lisp
(use-package treemacs
  :ensure t
  :commands (treemacs treemacs-select-window treemacs-find-file
             treemacs-add-and-display-current-project-exclusively)
  :custom
  ;; State, not cache: the workspace layout is worth keeping between sessions.
  (treemacs-persist-file (expand-file-name "treemacs/persist" kittymacs-etc-dir))
  (treemacs-last-error-persist-file
   (expand-file-name "treemacs/persist-at-last-error" kittymacs-cache-dir))
  (treemacs-width 35)
  (treemacs-width-is-initially-locked t)
  (treemacs-position 'left)
  (treemacs-display-in-side-window t)
  (treemacs-is-never-other-window t)
  (treemacs-follow-after-init t)
  (treemacs-file-follow-delay 0.2)
  (treemacs-project-follow-cleanup t)
  (treemacs-collapse-dirs (if (executable-find "python3") 3 0))
  (treemacs-indent-guide-style 'line)
  (treemacs-show-hidden-files t)
  (treemacs-hide-dot-git-directory t)
  (treemacs-silent-refresh t)
  (treemacs-silent-filewatch t)
  (treemacs-sorting 'alphabetic-asc)
  (treemacs-eldoc-display 'simple)
  (treemacs-no-png-images (not (display-graphic-p)))
  :config
  ;; Watch the filesystem, mark the current file in the fringe, draw the
  ;; indent guides, and show the project you are in and only that project.
  (treemacs-filewatch-mode 1)
  (treemacs-follow-mode 1)
  (treemacs-project-follow-mode 1)
  (treemacs-indent-guide-mode 1)
  (treemacs-fringe-indicator-mode 'always)

  ;; Git status colours.  The extended variants colour directories as well as
  ;; files and parse the status asynchronously in Python; `deferred' renders
  ;; the tree first and colours it a moment later, which feels quicker on a
  ;; large repository.  Without Python 3, the simple variant colours files.
  (pcase (cons (and (executable-find "git") t)
               (and treemacs-python-executable t))
    ('(t . t) (treemacs-git-mode 'deferred))
    ('(t . nil) (treemacs-git-mode 'simple)))
  ;; Show files that Git ignores, too.
  (treemacs-hide-gitignored-files-mode -1)

  ;; The sidebar is a panel that serves the buffer beside it; dimming it as
  ;; an "inactive window" only makes the tree harder to read.
  (with-eval-after-load 'dimmer
    (add-to-list 'dimmer-buffer-exclusion-regexps "\\*Treemacs")))

#+end_src

* Every file on disk is shown :decision:
:PROPERTIES:
:ID: c0abb9db-6cb2-4eee-ae75-343666cf53c6
:END:

- The tree hides only the =.git= directory itself: dotfiles are shown, and so are files Git ignores, such as build output or a local =private.el=.
  + Silently leaving out files that exist on disk confuses more than showing a few extra.
- =treemacs-hide-gitignored-files-mode= could hide ignored files; it is off by default, and switched off explicitly in [[*The sidebar]] so the choice is written down.
  + Note the =-1=: a minor mode called from Lisp with =nil= is turned /on/, not off.
- =SPC m h= toggles dotfiles for the moment.

* Icons, workspaces and Magit

- Three small packages connect Treemacs to what this configuration already runs.
- /Nerd Font icons/ replace the default PNG icons, so the tree matches the completion, Corfu and Dired icons.
  + Same check as [[id:0286f652-e861-4e61-9ca6-91e225c8939d][the appearance chapter]], so a machine without the font gets the plain theme instead of boxes.
- /Workspaces/: [[id:e4d65984-892d-4046-b718-d4265ab74215][the navigation chapter]] gives each tab its own buffers, and =treemacs-tab-bar= its own tree to match.
  + Switching workspace switches the sidebar with it.
- /Magit/ closes the one gap in filesystem watching.
  + Staging and committing change Git's index, not your files, so Treemacs would not otherwise notice the colours changed.

#+begin_src emacs-lisp
(use-package treemacs-nerd-icons
  :ensure t
  :after treemacs
  :demand t
  :config
  (when (kittymacs-icons-available-p)
    (treemacs-load-theme "nerd-icons")))

(use-package treemacs-tab-bar
  :ensure t
  :after treemacs
  :demand t
  :config
  (treemacs-set-scope-type 'Tabs))

(use-package treemacs-magit
  :ensure t
  :after (treemacs magit)
  :demand t)

#+end_src

* Meow in the tree :meow:
:PROPERTIES:
:ID: 7826ac9a-cde0-41ee-837c-3f54fbeaa3b7
:END:

- The tree is an application buffer, so it starts in /Motion/ state.
  + Treemacs's own keys are the interface; Meow adds only =j=/=k= and the leader.
  + Treemacs already reads =h= and =l= as collapse and expand, the same shape as [[id:878a9e46-62be-4b9f-82f6-589560d2ebc1][Dired]] here.
- =SPC m= labels what a new user looks for:

| Keys        | Does                                                                     |
|-------------+--------------------------------------------------------------------------|
| =f= =d=     | new file, new directory                                                  |
| =r= =D=     | rename, delete                                                           |
| =y= =m=     | copy, move                                                               |
| =p= =P=     | copy the absolute or relative path                                       |
| =o= =x=     | open in a chosen window (a directory opens in Dired), open outside Emacs |
| =g= =w= =h= | refresh, set width, show hidden files                                    |
| =b= =!=     | bookmark, shell command here                                             |
| =a= =k= =u= | add a project, remove a project, parent directory                        |
| =q=         | close the tree                                                           |
| =?=         | Treemacs's own help: every key it has                                    |

- =SPC m ?= follows the convention =?= has in every other mode menu here.

#+begin_src emacs-lisp
(with-eval-after-load 'meow
  (add-to-list 'meow-mode-state-list '(treemacs-mode . motion)))

(kittymacs-define-localleader 'treemacs-mode
  "f" (cons "new file" #'treemacs-create-file)
  "d" (cons "new directory" #'treemacs-create-dir)
  "r" (cons "rename" #'treemacs-rename-file)
  "D" (cons "delete" #'treemacs-delete-file)
  "y" (cons "copy" #'treemacs-copy-file)
  "m" (cons "move" #'treemacs-move-file)
  "p" (cons "copy path" #'treemacs-copy-absolute-path-at-point)
  "P" (cons "copy relative path" #'treemacs-copy-relative-path-at-point)
  "o" (cons "open elsewhere" #'treemacs-visit-node-ace)
  "x" (cons "open outside Emacs" #'treemacs-visit-node-in-external-application)
  "g" (cons "refresh" #'treemacs-refresh)
  "w" (cons "set width" #'treemacs-set-width)
  "h" (cons "show hidden files" #'treemacs-toggle-show-dotfiles)
  "b" (cons "bookmark" #'treemacs-add-bookmark)
  "!" (cons "shell command here" #'treemacs-run-shell-command-for-current-node)
  "a" (cons "add project" #'treemacs-add-project-to-workspace)
  "k" (cons "remove project" #'treemacs-remove-project-from-workspace)
  "u" (cons "parent directory" #'treemacs-root-up)
  "q" (cons "close the tree" #'treemacs-quit)
  "?" (cons "menu" #'treemacs-common-helpful-hydra))

(provide 'kittymacs-treemacs)
;;; kittymacs-treemacs.el ends here
#+end_src

* Check

1. =SPC t d= in any file of a project: =j= and =k= move, =TAB= opens a directory, =RET= opens a file in the window beside the tree.
2. Open a file from a different project (=SPC p p=): the tree follows.
3. Edit a file and save it: its name changes colour.
4. Stage it with =SPC v s= then =s=: the colour changes again without a refresh.
5. =SPC f t= from any buffer points the tree at the file you are editing.
6. =SPC m ?= lists every key Treemacs has, in a menu.

- =just test-packages= replays a real startup: this module must load and every =SPC= key must run a command.
