#+TITLE: Dashboard: home and discovery
#+FILETAGS: :interface:
#+STARTUP: overview
#+PROPERTY: header-args:emacs-lisp :tangle ../lisp/kittymacs-dashboard.el :eval never :comments no :padline no :mkdirp yes

- A home page, =*home*=: the dashboard package presenting Emacs's own recent files, =project.el= projects and bookmarks.
- kittymacs adds the banner and layout, buttons to the landing pages (the cheat sheet, the reading guide, the leader tree, Meow's tutor), a guide to the first keys, font-aware centring and the literate Config button.
- No second project or workspace database: the lists come from what Emacs already keeps.
- Leans on [[id:0286f652-e861-4e61-9ca6-91e225c8939d][the appearance chapter]] for fonts and icons and on [[id:53ad53ed-155f-47dc-973d-7be18de2893c][the keys chapter]] for the =SPC= keys it advertises.

* The cheat sheet, offline

- The module requires =kittymacs-ui= first: the page measures its text to centre it, so the fonts must already be chosen.
- The cheat sheet is an Org file in =lisp/= (=keybindings.org=), so it opens without a network.
- One command opens it from three places:
  + =?= on the home page;
  + =SPC h ?= anywhere else;
  + the "Keys & commands" button.

* Buttons that run commands

- Each button runs a command: open a file, switch project, open a recent file, the agenda, the scratch buffer, the reading guide, the cheat sheet, Meow's tutorial, and [[id:53ad53ed-155f-47dc-973d-7be18de2893c][the keys chapter]], which lists the whole =SPC= tree.
- Dashboard calls a button's action with the button itself as its argument.
  + A command like =find-file= cannot be the action directly: it would take the button for a file name.
  + =kittymacs--dashboard-run= wraps a command in a function that ignores the argument and runs the command as =M-x= would.
- The cheat-sheet and keys-chapter commands written here, and the literate =kittymacs-literate-open=, accept and ignore the argument themselves, so they need no wrapper.

#+begin_src emacs-lisp
(defun kittymacs--dashboard-run (command)
  "Return a dashboard button action that runs COMMAND interactively."
  (lambda (&rest _) (call-interactively command)))

(defun kittymacs-dashboard-open-keys-chapter (&rest _)
  "Open the keys chapter: the whole SPC tree, group by group."
  (interactive)
  (find-file (expand-file-name "literate/42-keys.org" user-emacs-directory)))
#+end_src

* Centring what is rendered, not characters :concept:
:PROPERTIES:
:ID: d9285bd0-2543-40af-ac5f-f09086967bf8
:END:

- Icons, proportional headings and display overlays have different pixel widths, so counting characters centres badly.
- The centring pass measures each actual buffer line instead:
  1. skip leading and trailing blanks, so trailing padding does not count;
  2. measure the visible span and the indentation before it with =window-text-pixel-size=;
  3. give the line a display-space prefix aligned to =(- center offset)=.
- The prefix is relative to the window centre, so it follows window resizing without remeasuring.
- During =after-init= the buffer may not be displayed yet, so the pass shows it in the selected window while it measures.

#+begin_src emacs-lisp
(defun kittymacs--dashboard-center-lines ()
  "Center each visible dashboard line using its rendered pixel width.
Measure the actual buffer so heading display overlays and icon faces count.
Exclude trailing padding and compensate for leading indentation."
  (let ((inhibit-read-only t)
        (inhibit-redisplay t)
        (buffer (current-buffer)))
    (save-window-excursion
      ;; During after-init this buffer may not have been displayed yet.
      (set-window-buffer (selected-window) buffer)
      (with-current-buffer buffer
        (remove-text-properties (point-min) (point-max)
                                '(line-prefix nil wrap-prefix nil indent-prefix nil))
        (save-excursion
          (goto-char (point-min))
          (while (not (eobp))
            (let* ((start (line-beginning-position))
                   (end (line-end-position))
                   (first (progn (skip-chars-forward " \t" end) (point)))
                   (last (save-excursion
                           (goto-char end)
                           (skip-chars-backward " \t" start)
                           (point))))
              (when (< first last)
                (let* ((width (car (window-text-pixel-size
                                    (selected-window) start last t)))
                       (indent (car (window-text-pixel-size
                                     (selected-window) start first t)))
                       (offset (/ (+ width indent) 2.0))
                       (prefix (propertize
                                " " 'display
                                `(space :align-to (- center (,offset))))))
                  (add-text-properties start end
                                       `(line-prefix ,prefix wrap-prefix ,prefix)))))
            (forward-line)))))))
#+end_src

- Font and theme changes do change widths, so they trigger a fresh measurement.
  + Only a visible home page is measured.
  + The font, theme and initial window-setup hooks call this after their changes settle ([[*The home page]]).

#+begin_src emacs-lisp
(defun kittymacs--dashboard-recenter (&rest _)
  "Recompute visible dashboard text metrics after a font or theme change."
  (when-let* ((window (get-buffer-window dashboard-buffer-name t)))
    (with-selected-window window
      (with-current-buffer dashboard-buffer-name
        (kittymacs--dashboard-center-lines)))))
#+end_src

* A guide to the first keys

- The page ends with the six keys the README and the cheat sheet teach first, in two columns, so a fresh install says where to go before anything is opened.

| Key       | Does                  | Key       | Does              |
|-----------+-----------------------+-----------+-------------------|
| =SPC SPC= | run a command by name | =SPC h ?= | the cheat sheet   |
| =SPC f f= | open a file           | =SPC h t= | Meow's tutorial   |
| =SPC m=   | menu for this mode    | =SPC C c= | the reading guide |

- Keys use the navigator face, descriptions the comment face; the centring pass aligns each line like the rest of the page.
- The rows are data: adding one is adding a row to =kittymacs-dashboard-key-guide=.

#+begin_src emacs-lisp
(defvar kittymacs-dashboard-key-guide
  '(("SPC SPC" "run a command by name" "SPC h ?" "the cheat sheet")
    ("SPC f f" "open a file"           "SPC h t" "Meow's tutorial")
    ("SPC m"   "menu for this mode"    "SPC C c" "the reading guide"))
  "Rows of (KEY WHAT KEY WHAT) shown at the bottom of the home page.")

(defun kittymacs--dashboard-insert-key-guide ()
  "Insert the short guide to the first keys."
  (insert "\n")
  (dolist (row kittymacs-dashboard-key-guide)
    (pcase-let ((`(,left-key ,left-what ,right-key ,right-what) row))
      (insert (propertize (format "%-8s" left-key) 'face 'dashboard-navigator)
              (propertize (format "%-24s" left-what) 'face 'font-lock-comment-face)
              (propertize (format "%-8s" right-key) 'face 'dashboard-navigator)
              (propertize right-what 'face 'font-lock-comment-face)
              "\n"))))
#+end_src

* The home page

- Everything the page shows is a dashboard option, set with =:custom= like any other package's:
  + the kittymacs banner and the three button rows;
  + bounded lists from Emacs's own recent files (5), projects (5) and bookmarks (3), on =r=, =p= and =b=;
  + the init line (Emacs version and start-up time) and the key guide.
- Icons appear only when =kittymacs-icons-available-p= says the font has them.
- The last item of =dashboard-startupify-list= is the centring pass, so it measures the finished page.
- =?= opens the cheat sheet from the home page.
- The recentring hook runs at depth 100 on =window-setup-hook=, =after-setting-font-hook= and =enable-theme-functions=.
- The page is skipped when Emacs was started with a file to open.

#+begin_src emacs-lisp
(use-package dashboard
  :ensure t
  :demand t
  :custom
  (dashboard-buffer-name "*home*")
  (dashboard-startup-banner 'ascii)
  (dashboard-banner-ascii
   (mapconcat #'identity
              '("╭────────────────────────────╮"
                "│                            │"
                "│   k i t t y m a c s   :3   │"
                "│                            │"
                "╰────────────────────────────╯")
              "\n"))
  (dashboard-banner-logo-title "select · extend · act")
  (dashboard-center-content t)
  (dashboard-vertically-center-content nil)
  (dashboard-navigation-cycle t)
  (dashboard-hide-cursor t)
  (dashboard-icon-type 'nerd-icons)
  (dashboard-set-heading-icons t)
  (dashboard-set-file-icons t)
  (dashboard-display-icons-p #'kittymacs-icons-available-p)
  (dashboard-heading-icon-height 1.0)
  (dashboard-show-shortcuts t)
  (dashboard-projects-backend 'project-el)
  (dashboard-path-style 'truncate-middle)
  (dashboard-path-max-length 48)
  (dashboard-items '((recents . 5) (projects . 5) (bookmarks . 3)))
  (dashboard-item-shortcuts '((recents . "r") (projects . "p") (bookmarks . "b")))
  (dashboard-item-names '(("Recent Files:" . "Recent")
                          ("Projects:" . "Projects")
                          ("Bookmarks:" . "Bookmarks")))
  (dashboard-startupify-list '(dashboard-insert-banner
                               dashboard-insert-banner-title
                               dashboard-insert-newline
                               dashboard-insert-navigator
                               dashboard-insert-newline
                               dashboard-insert-init-info
                               dashboard-insert-items
                               kittymacs--dashboard-insert-key-guide
                               kittymacs--dashboard-center-lines))
  (dashboard-init-info (lambda ()
                         (format "Emacs %s · ready in %s" emacs-version (emacs-init-time))))
  (dashboard-navigator-buttons
   `((("+" "File" "Open a file" ,(kittymacs--dashboard-run #'find-file))
      ("◆" "Project" "Switch project" ,(kittymacs--dashboard-run #'project-switch-project))
      ("↺" "Recent" "Open a recent file" ,(kittymacs--dashboard-run #'consult-recent-file)))
     (("◎" "Agenda" "Open the Org agenda" ,(kittymacs--dashboard-run #'org-agenda))
      ("*" "Scratch" "Open the scratch buffer" ,(kittymacs--dashboard-run #'scratch-buffer))
      ("λ" "Config" "Open the kittymacs reading guide (SPC C c)" kittymacs-literate-open))
     (("?" "Keys & commands" "Open the local cheat sheet (SPC h ?, or ? here)"
       kittymacs-dashboard-open-cheatsheet)
      ("»" "Meow tutor" "Learn select, extend, act (SPC h t)"
       ,(kittymacs--dashboard-run #'meow-tutor))
      ("§" "Leader tree" "Every SPC key, group by group"
       kittymacs-dashboard-open-keys-chapter))))
  :config
  (set-face-attribute 'dashboard-text-banner nil
                      :inherit 'font-lock-keyword-face
                      :weight 'bold)
  (set-face-attribute 'dashboard-banner-logo-title nil
                      :inherit 'font-lock-comment-face
                      :height 1.05)
  (set-face-attribute 'dashboard-heading nil
                      :inherit 'font-lock-function-name-face
                      :weight 'bold)
  (set-face-attribute 'dashboard-navigator nil
                      :inherit 'font-lock-keyword-face
                      :weight 'semi-bold)

  (keymap-set dashboard-mode-map "?" #'kittymacs-dashboard-open-cheatsheet)

  (add-hook 'window-setup-hook #'kittymacs--dashboard-recenter 100)
  (add-hook 'after-setting-font-hook #'kittymacs--dashboard-recenter 100)
  (add-hook 'enable-theme-functions #'kittymacs--dashboard-recenter 100)

  ;; Skip the home page when Emacs was invoked with a file argument.
  (dashboard-setup-startup-hook))
#+end_src

* Meow on the home page :meow:
:PROPERTIES:
:ID: 545c730c-f8a5-4b01-bb3a-c39a4f04ce0c
:END:

- The home page starts in Motion state.
  + The page's own keys (=r=, =p=, =b=, =?= and the item shortcuts) keep working alongside Meow's =j=/=k= and =SPC=.
- No =SPC m= menu: the page's keys are already single letters.
- The setting waits for Meow to load rather than racing its initialization, and does not force Meow to load early.

#+begin_src emacs-lisp
;; Keep r/p/b/? and dashboard item shortcuts alongside Meow j/k and SPC.
(with-eval-after-load 'meow
  (add-to-list 'meow-mode-state-list '(dashboard-mode . motion)))
#+end_src

* Check

- Start Emacs with no file argument: =*home*= shows the banner, three button rows, the lists and the key guide, centred.
- Start it with a file: the file opens and the home page does not.
- =?= on the home page opens the cheat sheet; =j=/=k= move; =r=, =p=, =b= jump to a list.
- Change the font size or theme: the page recentres.
- =SPC o h= reopens the home page.
- =just tangle-check= confirms =lisp/kittymacs-dashboard.el= matches this page.
