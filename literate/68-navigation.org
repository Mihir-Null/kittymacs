#+TITLE: Getting around: projects, recent files, search and workspaces
#+FILETAGS: :apps:
#+STARTUP: overview
#+PROPERTY: header-args:emacs-lisp :tangle ../lisp/kittymacs-navigation.el :eval never :comments no :padline no :mkdirp yes

- Getting around files, projects and workspaces:
  + Emacs remembers where you were in every file and which files you opened recently (=SPC f r=);
  + a /project/ is any Git repository or directory with a marker file; =SPC p= finds files, buffers and text inside one;
  + a /workspace/ is a tab with its own set of buffers, usually one per project, under =SPC W=;
  + searches use ripgrep when it is installed.
- Most of it is built into Emacs (=project.el=, =recentf=, =saveplace=, bookmarks, the tab bar); packages add jumping (Avy), live replacement (=visual-regexp=) and workspaces (Tabspaces).
- Leans on [[id:42b02bb4-ea68-4981-b427-35705bf162f4][the completion chapter]] for Consult, [[id:3fd783cd-deb8-48aa-bb44-0d9b16535365][the defaults chapter]] for the cache directory and [[id:afa6b7fd-742a-4c15-86a6-32e3e8027b0e][the platform chapter]], and on [[id:e5942eb0-4389-447d-9317-d9178adcaaae][the leader]] for the groups its keys go in.
- Distilled from Lambda-Emacs by Colin McLear (=lem-setup-navigation=, =-projects=, =-search=, =-tabs=).
- Those three are the =jump=, =visual-replace= and =workspaces= features, each behind a =when= with its keys.
  + Without =workspaces=, =SPC p p= is Emacs's own =project-switch-project=, and =SPC W= still holds the plain tab commands.

* Remembering places and files

- =save-place-mode= reopens every file where you left it, and keeps the place of a file it cannot read right now.
- =recentf= keeps the last 500 files, for =SPC f r= and the home page.
- Places, recent files and bookmarks are saved under the cache directory.
- URLs and e-mail addresses are clickable in compilation, shell, Eshell and text buffers, and in the comments and strings of code.
- =SPC b j= jumps to a heading or definition with completion: Org headings in Org, the outline elsewhere.

  "Jump to a heading or definition in this buffer with completion."
  (interactive)

      (call-interactively #'consult-org-heading)

* Jump to anything on screen

- Avy overlays a letter on every match of what you type; press the letter and point is there.
- It works across every window of the frame; a pause of 0.4 s ends the input.

| Key       | Jumps to                    |
|-----------+-----------------------------|
| =SPC j j= | a character or two you type |
| =SPC j l= | a line                      |
| =SPC j w= | a word                      |
| =SPC j s= | a symbol                    |
| =SPC j e= | the end of a line           |
| =SPC j i= | a character in this line    |

#+begin_src emacs-lisp
(when (kittymacs-feature-p 'jump)
  (use-package avy
    :ensure t
    :commands (avy-goto-char-timer avy-goto-line avy-goto-word-1 avy-goto-symbol-1
               avy-goto-end-of-line avy-goto-char-in-line)
    :custom
    (avy-timeout-seconds 0.4)
    (avy-all-windows t)
    (avy-style 'at-full))

  (keymap-set kittymacs-jump-map "j" (cons "to character" #'avy-goto-char-timer))
  (keymap-set kittymacs-jump-map "l" (cons "to line" #'avy-goto-line))
  (keymap-set kittymacs-jump-map "w" (cons "to word" #'avy-goto-word-1))
  (keymap-set kittymacs-jump-map "s" (cons "to symbol" #'avy-goto-symbol-1))
  (keymap-set kittymacs-jump-map "e" (cons "to end of line" #'avy-goto-end-of-line))
  (keymap-set kittymacs-jump-map "i" (cons "in this line" #'avy-goto-char-in-line)))

* Projects

- =project.el= is Emacs's project layer; the project list is saved under the cache.
- Switching to a project asks what to do there: find a file, a regexp or a directory, VC-Dir, or Magit on =G=.
  + A command in that menu names its key when =project-prefix-map= has no key for it; [[id:e5942eb0-4389-447d-9317-d9178adcaaae][the leader]] leaves that map as Emacs defines it.
- Extra root markers let non-Git directories count as projects: =.dir-locals.el=, =.project.el=, =package.json=, =requirements.txt=, =autogen.sh=.
- Ripgrep backs =xref= searches when =rg= is installed.
- Projects whose directories no longer exist are forgotten when =project.el= loads.
- =SPC p t= opens the directory where projects live (=kittymacs-project-directory=).
- =SPC p R= remembers every project under a directory you name.

#+begin_src emacs-lisp
(defun kittymacs-projects-directory ()
  "Open the directory where projects live."
  (interactive)
  (dired kittymacs-project-directory))

(keymap-set kittymacs-project-map "t" (cons "projects directory" #'kittymacs-projects-directory))
(keymap-set kittymacs-project-map "R" (cons "remember projects under" #'project-remember-projects-under))

(use-package project
  :ensure nil
  :custom
  (project-list-file (expand-file-name "projects" kittymacs-cache-dir))
  (project-switch-commands '((project-find-file "Find file")
                             (project-find-regexp "Find regexp")
                             (project-find-dir "Find directory")
                             (project-vc-dir "VC-Dir")
                             (magit-project-status "Magit status" ?G)))
  (project-vc-extra-root-markers '(".dir-locals.el" ".project.el" "package.json" "requirements.txt" "autogen.sh"))
  :config
  (when (executable-find "rg")
    (setopt xref-search-program 'ripgrep))
  (project-forget-zombie-projects))

#+end_src

* Workspaces: tabs with their own buffers :decision:
:PROPERTIES:
:ID: 826962f8-d451-4444-8e0e-d0f0024da8b6
:END:

- The tab bar holds window configurations.
  + Tabs show their numbers; a new tab opens rightmost on the scratch buffer.
  + No new-tab or close buttons; closing the last tab turns the tab bar off.
- =SPC TAB= (=kittymacs-tab-dwim=) creates a tab when there is none, switches when there are two, and otherwise asks which.
  + =SPC W TAB= is the same command, and =SPC {= and =SPC }= move between tabs.
- Tabspaces turns each tab into a workspace whose buffer list shows only the buffers opened inside it; the first is called "Home".
  + =SPC b b= in a project's tab lists that project's buffers first.
  + The full list is a narrow key away; =w= narrows back to the workspace.
  + Turning Tabspaces off restores Consult's ordinary buffer list.
  + =SPC p p= opens a project in its own workspace; =C-x p p= still switches project in the current one.
- Tabspaces is one of two global modes this configuration does not switch on in =:config= (envrc is the other); it turns on from =after-init-hook= at depth 95, through =kittymacs-after-init=, so a failure there is a warning rather than the end of startup.
  + The mode reads some of its options only when it turns on:
    - whether =switch-to-buffer= shows the workspace's buffers (=tabspaces-use-filtered-buffers-as-default=);
    - whether sessions are saved and restored (=tabspaces-session=, =tabspaces-session-auto-restore=);
    - whether the echo area names the workspace (=tabspaces-echo-area-enable=).
  + Changing one of them later does nothing until the mode is turned off and on again.
  + Depth 95 is after the Customize file and after the =private.el= function at depth 90 ("Machine-specific settings" in [[id:40302e33-d2b2-4c0b-9bc3-1f9e121307e1][the startup chapter]]), so a value set in either place is the one it reads.

(defun kittymacs-tab-dwim ()
  "Create a tab if there is one, switch if there are two, else choose one."
  (interactive)
  (let ((tabs (mapcar (lambda (tab) (alist-get 'name tab)) (tab-bar--tabs-recent))))
    (cond ((null tabs) (tab-new))
          ((= (length tabs) 1) (tab-next))
          (t (tab-bar-switch-to-tab (completing-read "Select tab: " tabs nil t))))))

(keymap-set kittymacs-leader-map "{" (cons "previous tab" #'tab-bar-switch-to-prev-tab))
(keymap-set kittymacs-leader-map "}" (cons "next tab" #'tab-bar-switch-to-next-tab))
(keymap-set kittymacs-leader-map "TAB" (cons "switch tab" #'kittymacs-tab-dwim))

(keymap-set kittymacs-workspace-map "TAB" (cons "switch" #'kittymacs-tab-dwim))
(keymap-set kittymacs-workspace-map "n" (cons "new tab" #'tab-new))
(keymap-set kittymacs-workspace-map "]" (cons "next tab" #'tab-next))
(keymap-set kittymacs-workspace-map "[" (cons "previous tab" #'tab-previous))

(defvar kittymacs-consult-source-workspace
  (list :name "Workspace Buffers"
        :narrow ?w
        :history 'buffer-name-history
        :category 'buffer
        :state #'consult--buffer-state
        :default t
        :items (lambda () (consult--buffer-query
                           :predicate #'tabspaces--local-buffer-p
                           :sort 'visibility
                           :as #'buffer-name)))
  "Consult source listing only this workspace's buffers.")

(defun kittymacs--consult-tabspaces ()
  "Show workspace buffers first while tabspaces is on."
  (if tabspaces-mode
      (progn
        (plist-put consult-source-buffer :hidden t)
        (plist-put consult-source-buffer :default nil)
        (add-to-list 'consult-buffer-sources 'kittymacs-consult-source-workspace))
    (plist-put consult-source-buffer :hidden nil)
    (plist-put consult-source-buffer :default t)
    (setq consult-buffer-sources (remove 'kittymacs-consult-source-workspace consult-buffer-sources))))

(when (kittymacs-feature-p 'workspaces)
  (use-package tabspaces
    :ensure t
    :demand t
    :custom
    (tabspaces-use-filtered-buffers-as-default t)
    (tabspaces-default-tab "Home")
    :config
    (kittymacs-after-init #'tabspaces-mode 95)
    (with-eval-after-load 'consult
      (add-hook 'tabspaces-mode-hook #'kittymacs--consult-tabspaces)
      (kittymacs--consult-tabspaces))))

* Check

- =SPC p p= opens a project in its own workspace; =SPC b b= now lists only its buffers.
- =SPC s d= then a word: ripgrep across the project, with live preview.
- =SPC f r= for recent files; reopen one and you are where you left off.
- =SPC j j= then a character: letters appear on every match in the frame.
- =C-h v tabspaces-mode=: =t= once started, after the =private.el= function has run.
- =just test-packages=: the startup verifier replays a real startup, after-init hooks included, and fails on any warning.
